# Drum Something's MIDI: a device of its own beside the synth (GLO DRUMS): its note channel and its CC channel,
# apart from GLO MIDI's; the 400's key map on its channel; its CCs (33-45 each drum's tuning, 46-58 its volume);
# both kept across a reboot. tools/build.py bs2 --host, then from the repository's root
#   bs2/build/host/bass-something-2_host bs2/tests/drums.txt bs2/build/host
wait 600
listen
# GLO: MIDI, again DRUMS
tap GLO
wait 30
tap GLO
wait 30
expect page 24
said GLOBAL DRUMS
# the defaults: the drums on channel 10 (GM's), their CCs too; the synth on 1
expect dchan 10
expect dcc_chan 10
expect chan 1
# channel 10: the drums play (the snare, 38), the synth does not
midi 99 26 7F
wait 50
expect dt_hits 1
expect note -1
# channel 1: the synth plays, the drums do not
midi 90 3C 64
wait 50
expect note 60
expect dt_hits 1
midi 80 3C 00
wait 50
# the drums on channel 1 too: one note on it plays both, as two devices on one channel
spin K1 -9
wait 30
expect dchan 1
said DRUMS CHANNEL 1
midi 90 26 64
wait 50
expect note 38
expect dt_hits 2
midi 80 26 00
wait 50
# a note off: nothing for the drums (the 400 ignored them)
expect dt_hits 2
# OFF: the drums hear no notes
spin K1 -1
wait 30
expect dchan 0
said DRUMS CHANNEL OFF
midi 99 26 7F
wait 50
expect dt_hits 2
# ALL: the drums hear every channel
spin K1 -1
wait 30
expect dchan 0
spin K1 17
wait 30
expect dchan 17
said DRUMS CHANNEL ALL
midi 95 26 7F
wait 50
expect dt_hits 3
# back to 10
spin K1 -7
wait 30
expect dchan 10
# their CCs on channel 10: CC 34 the snare's tuning (64: step 8 of 0-15), CC 47 its volume (0)
midi B9 22 40
wait 30
expect dtune1 8
midi B9 2F 00
wait 30
expect dvol1 0
midi B9 2F 7F
wait 30
expect dvol1 15
# the synth's CCs are not the drums': CC 34 on channel 1 leaves the snare's tuning
midi B0 22 00
wait 30
expect dtune1 8
# CC CHAN 9 (SELECT, on PRESET as GLO MIDI's): CCs on 10 no longer reach the drums
spin PRESET -1
wait 30
expect dcc_chan 9
said DRUMS CC CHANNEL 9
midi B9 22 00
wait 30
expect dtune1 8
midi B8 22 00
wait 30
expect dtune1 0
# SAVE keeps them: a reboot reads them back
tap SAVE
wait 30
reboot
wait 600
expect dchan 10
expect dcc_chan 9
