# The USB host (GLO CONNECT's USB HOST, K1 on K2): a USB MIDI keyboard on the port instead of a computer, its notes
# played, the FM-1's going to it; unplugged, its held note let go; off again, the computer back; off at every start.
# tools/build.py bs2 --host, then from the repository's root
#   bs2/build/host/bass-something-2_host bs2/tests/usbhost.txt bs2/build/host
wait 600
listen
tap GLO
wait 30
tap GLO
wait 30
expect page 24
said GLOBAL CONNECT
expect usb_host 0
expect usbh 0
spin K2 1
wait 30
expect usb_host 1
said USB HOST LOOKING
expect usbh 1
usbh plug Launchkey_Mini_MK3
wait 300
expect usbh 2
said USB DEVICE Launchkey Mini MK3
# its notes play the bass; the keys' go out to it
usbh midi 90 40 64
wait 30
expect note 64
out
key 7 down
wait 50
key 7 up
wait 50
expect_out USBH 90 3C 64
# unplugged with a note held: let go
usbh midi 90 43 64
wait 30
expect note 67
usbh unplug
wait 50
expect note -1
said USB DEVICE LOST
expect usbh 1
# off: a device again, the computer's
spin K2 -1
wait 30
expect usb_host 0
expect usbh 0
said USB HOST OFF
# on, then a restart: off (a device at every start)
spin K2 1
wait 2500
reboot
wait 600
expect usb_host 0
